\begin{lexlean}{Studio.Feedback.Masking}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\importmodule{Studio.Feedback.Intake}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: maskFrom,
  parameters: [text : Bytes, player : Bytes, offset : Nat, remaining : Nat, out : Bytes],
  result: Bytes,
  body: (match remaining
    | Nat.zero => out
    | Nat.succ rest => (if ((prim length text : Nat) <=? offset) then out else (match (prim slice text offset (prim length player : Nat) : (Option Bytes))
      | Option.some piece => (if (Studio.Feedback.Intake::same piece player) then (maskFrom text player (offset + (prim length player : Nat)) rest (prim append out 0x2a : Bytes)) else (maskFrom text player (offset + 1) rest (prim append out (Studio.Feedback.Intake::pieceAt text offset (offset + 1)) : Bytes)))
      | Option.none => (maskFrom text player (offset + 1) rest (prim append out (Studio.Feedback.Intake::pieceAt text offset (offset + 1)) : Bytes))))),
  recursive_argument: remaining,
  axioms: [Quot.sound, propext]
}

definition{
  name: mask,
  parameters: [text : Bytes, player : Bytes],
  result: Bytes,
  body: (if ((prim length player : Nat) =? 0) then text else (maskFrom text player 0 (prim length text : Nat) 0x))
}

definition{
  name: maskedText,
  parameters: [form : Studio.Feedback.Intake::Form],
  result: Bytes,
  body: (mask (project form text) (project form player))
}

theorem{
  name: masks_every_mention_of_the_player,
  parameters: [],
  statement: ((mask 0x70372063726173686564207768656e207037206a756d706564 0x7037) = 0x2a2063726173686564207768656e202a206a756d706564),
  proof: decide{},

}

  parameters: [],
  statement: ((mask 0x6e6f20736f756e64 0x7037) = 0x6e6f20736f756e64),
  proof: decide{},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
