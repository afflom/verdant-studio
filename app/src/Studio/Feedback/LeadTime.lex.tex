definition{
  name: inOrder,
  parameters: [received : Nat, published : Nat],
  result: Bool,
  body: (received <=? published)
}

definition{
  name: elapsed,
  parameters: [received : Nat, published : Nat],
  result: Nat,
  body: (prim subtract published received : Nat)
}

definition{
  name: leadTime,
  parameters: [received : Nat, published : Nat],
  result: (Option Nat),
  body: (if (inOrder received published) then (ctor Option.some [Nat] (elapsed received published)) else (ctor Option.none [Nat]))
}

definition{
  name: breaksLeadTime,
  parameters: [received : Nat, took : Nat],
  result: Bool,
  body: ((not (inOrder received (received + took))) || (not ((elapsed received (received + took)) =? took)))
}
