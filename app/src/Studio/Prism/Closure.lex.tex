definition{
  name: hasFrom,
  parameters: [set : Bytes, code : Nat, offset : Nat, remaining : Nat],
  result: Bool,

    | Nat.zero => false

  recursive_argument: remaining

definition{
  name: touchesFrom,

definition{
  name: reached,

  result: Bool,

definition{
  name: stepFrom,
  parameters: [pairs : Bytes, set : Bytes, offset : Nat, remaining : Nat, next : Bytes],
  result: Bytes,
  body: (match remaining
    | Nat.zero => next
    | Nat.succ rest => (if (reached pairs set (byteNat (letters) offset)) then (stepFrom pairs set (offset + 1) rest (prim append next (octetAt (letters) offset) : Bytes)) else (stepFrom pairs set (offset + 1) rest next))),
  recursive_argument: remaining
}

  name: closeFrom,

definition{
  name: sameFrom,
  parameters: [value : Bytes, pattern : Bytes, offset : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => true
    | Nat.succ rest => (((byteNat value offset) =? (byteNat pattern offset)) && (sameFrom value pattern (offset + 1) rest))),
  recursive_argument: remaining
}

definition{
  name: occursFrom,
  parameters: [text : Bytes, line : Bytes, offset : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => false
    | Nat.succ rest => ((match (prim slice text offset (prim length line : Nat) : (Option Bytes))
      | Option.none => false
      | Option.some part => (sameFrom part line 0 (prim length line : Nat))) || (occursFrom text line (offset + 1) rest))),
  recursive_argument: remaining
}

  result: Bytes,
  body: (match remaining
    | Nat.zero => out
    | Nat.succ rest => (if (hasCell table (octetAt domains offset) phase) then (spreadFrom table domains phase (offset + 1) rest (prim append out (prim append (prim append (prim append (octetAt domains offset) 0x2f : Bytes) phase : Bytes) 0x0a : Bytes) : Bytes)) else (spreadFrom table domains phase (offset + 1) rest out))),
  recursive_argument: remaining
}
