structure{
  name: Version,
  type_parameters: [],
  parameters: [],
  fields: [{name: major, type: Nat}, {name: minor, type: Nat}, {name: patch, type: Nat}]

definition{
  name: changePatch,
  parameters: [],
  result: Nat,
  body: 0
}

definition{
  name: changeMinor,
  parameters: [],
  result: Nat,
  body: 1
}

definition{
  name: changeMajor,
  parameters: [],
  result: Nat,
  body: 2
}

definition{
  name: nextMajor,
  parameters: [major : Nat, change : Nat],
  result: Nat,
  body: (if (change =? (changeMajor)) then (major + 1) else major)
}

definition{
  name: nextMinor,
  parameters: [minor : Nat, change : Nat],
  result: Nat,
  body: (if (change =? (changeMajor)) then 0 else (if (change =? (changeMinor)) then (minor + 1) else minor))
}

definition{
  name: nextPatch,
  parameters: [patch : Nat, change : Nat],
  result: Nat,
  body: (if (change =? (changeMajor)) then 0 else (if (change =? (changeMinor)) then 0 else (patch + 1)))
}

definition{
  name: next,
  parameters: [current : Version, change : Nat],
  result: Version,
  body: (record Version {major := (nextMajor (project current major) change), minor := (nextMinor (project current minor) change), patch := (nextPatch (project current patch) change)})
}

definition{
  name: precedesBy,
  parameters: [leftMajor : Nat, leftMinor : Nat, leftPatch : Nat, rightMajor : Nat, rightMinor : Nat, rightPatch : Nat],
  result: Bool,
  body: ((leftMajor <? rightMajor) || ((leftMajor =? rightMajor) && ((leftMinor <? rightMinor) || ((leftMinor =? rightMinor) && (leftPatch <? rightPatch)))))
}

definition{
  name: precedes,
  parameters: [left : Version, right : Version],
  result: Bool,
  body: (precedesBy (project left major) (project left minor) (project left patch) (project right major) (project right minor) (project right patch))
}

definition{
  name: breaksIncrease,
  parameters: [major : Nat, minor : Nat, patch : Nat, change : Nat],
  result: Bool,
  body: (not (precedesBy major minor patch (nextMajor major change) (nextMinor minor change) (nextPatch patch change)))
}

theorem{
  name: next_has_higher_precedence,
  parameters: [major : Nat, minor : Nat, patch : Nat, change : Nat],
  statement: ((breaksIncrease major minor patch change) = false),
  proof: linear_arithmetic{definitions: [breaksIncrease, changeMajor, changeMinor, nextMajor, nextMinor, nextPatch, precedesBy]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksReset,
  parameters: [minor : Nat, patch : Nat, change : Nat],
  result: Bool,
  body: (((change =? (changeMajor)) && (not (((nextMinor minor change) =? 0) && ((nextPatch patch change) =? 0)))) || ((change =? (changeMinor)) && (not ((nextPatch patch change) =? 0))))
}

theorem{
  name: next_resets_the_lower_numbers,
  parameters: [minor : Nat, patch : Nat, change : Nat],
  statement: ((breaksReset minor patch change) = false),
  proof: linear_arithmetic{definitions: [breaksReset, changeMajor, changeMinor, nextMinor, nextPatch]},
  axioms: [Quot.sound, propext]
}

definition{
  name: sameFrom,
  parameters: [left : Bytes, right : Bytes, offset : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => true
    | Nat.succ rest => (match (prim index left offset : (Option UInt8))
      | Option.none => true
      | Option.some l => (match (prim index right offset : (Option UInt8))
        | Option.none => false
        | Option.some r => (if ((prim to_nat l : Nat) =? (prim to_nat r : Nat)) then (sameFrom left right (offset + 1) rest) else false)))),
  recursive_argument: remaining,
  axioms: [Quot.sound, propext]
}

definition{
  name: same,
  parameters: [left : Bytes, right : Bytes],
  result: Bool,
  body: (((prim length left : Nat) =? (prim length right : Nat)) && (sameFrom left right 0 (prim length left : Nat))),
  axioms: [Quot.sound, propext]
}

definition{
  name: fieldAt,
  parameters: [value : Bytes, start : Nat, stop : Nat],
  result: Bytes,
  body: (match (Foundation.Bytes::sliceBytes value start (prim subtract stop start : Nat))
    | Option.none => 0x
    | Option.some piece => piece)
}

definition{
  name: fields,
  parameters: [value : Bytes, separator : Nat, offset : Nat, start : Nat, remaining : Nat],
  result: (List Bytes),
  body: (match remaining
    | Nat.zero => (cons (fieldAt value start offset) (nil Bytes))
    | Nat.succ rest => (match (prim index value offset : (Option UInt8))
      | Option.none => (cons (fieldAt value start offset) (nil Bytes))
      | Option.some raw => (if ((prim to_nat raw : Nat) =? separator) then (cons (fieldAt value start offset) (fields value separator (offset + 1) (offset + 1) rest)) else (fields value separator (offset + 1) start rest)))),
  recursive_argument: remaining,
  axioms: [Quot.sound, propext]
}

definition{
  name: split,
  parameters: [value : Bytes, separator : Nat],

definition{
  name: digitsFrom,
  parameters: [value : Bytes, offset : Nat, remaining : Nat, total : Nat],
  result: (Option Nat),
  body: (match remaining
    | Nat.zero => (ctor Option.some [Nat] total)
    | Nat.succ rest => (match (prim index value offset : (Option UInt8))
      | Option.none => (ctor Option.some [Nat] total)
      | Option.some raw => (if ((48 <=? (prim to_nat raw : Nat)) && ((prim to_nat raw : Nat) <=? 57)) then (digitsFrom value (offset + 1) rest ((prim multiply total 10 : Nat) + (prim subtract (prim to_nat raw : Nat) 48 : Nat))) else (ctor Option.none [Nat])))),
  recursive_argument: remaining,
  axioms: [Quot.sound, propext]
}

definition{
  name: leadingZero,
  parameters: [value : Bytes],
  result: Bool,
  body: ((1 <? (prim length value : Nat)) && (match (prim index value 0 : (Option UInt8))
    | Option.none => false
    | Option.some raw => ((prim to_nat raw : Nat) =? 48))),
  axioms: [Quot.sound, propext]
}

definition{
  name: number,
  parameters: [value : Bytes],
  result: (Option Nat),
  body: (if (((prim length value : Nat) =? 0) || (leadingZero value)) then (ctor Option.none [Nat]) else (digitsFrom value 0 (prim length value : Nat) 0)),
  axioms: [Quot.sound, propext]
}

definition{
  name: parseVersion,
  parameters: [value : Bytes],
  result: (Option Version),
  body: (match (split value 46)
    | List.nil => (ctor Option.none [Version])
    | List.cons x more => (match more
      | List.nil => (ctor Option.none [Version])
      | List.cons y more2 => (match more2
        | List.nil => (ctor Option.none [Version])
        | List.cons z more3 => (match more3
          | List.cons extra others => (ctor Option.none [Version])
          | List.nil => (match (number x)
            | Option.none => (ctor Option.none [Version])
            | Option.some majorNumber => (match (number y)
              | Option.none => (ctor Option.none [Version])
              | Option.some minorNumber => (match (number z)
                | Option.none => (ctor Option.none [Version])
                | Option.some patchNumber => (ctor Option.some [Version] (record Version {major := majorNumber, minor := minorNumber, patch := patchNumber}))))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: judge,
  parameters: [revision : Bytes, current : Bytes, increment : Bytes, gates : (List Bytes), fuel : Nat],
  result: Bytes,
  body: (match (parseVersion current)
    | Option.none => (versionRefusal current)
    | Option.some released => (match (parseChange increment)
      | Option.none => (changeRefusal increment)
      | Option.some step => (match gates
        | List.nil => (noGateRefusal revision)
        | List.cons first others => (match (firstFailed gates)
          | Option.some failed => (gateRefusal revision failed)
          | Option.none => (cut revision (next released step) fuel))))),
  axioms: [Quot.sound, propext]
}
