definition{
  name: octet,
  parameters: [value : Nat],
  result: Bytes,
  body: (match (prim slice (octets) (prim remainder value 256 0 : Nat) 1 : (Option Bytes))
    | Option.none => 0x
    | Option.some piece => piece)
}

definition{
  name: notesFrom,
  parameters: [features : Bytes, offset : Nat, remaining : Nat, out : Bytes],
  result: Bytes,
  body: (match remaining
    | Nat.zero => out
    | Nat.succ rest => (match (prim slice features offset 1 : (Option Bytes))
      | Option.none => out
      | Option.some key => (notesFrom features (offset + 1) rest (prim append out (note key) : Bytes)))),
  recursive_argument: remaining
}
