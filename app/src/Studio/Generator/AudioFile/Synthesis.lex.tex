definition{
  name: pairUp,
  parameters: [pieces : (List Bytes), fuel : Nat],
  result: (List Bytes),
  body: (match fuel
    | Nat.zero => pieces
    | Nat.succ rest => (match pieces
      | List.nil => (nil Bytes)
      | List.cons head others => (match others
        | List.nil => (cons head (nil Bytes))
        | List.cons following more => (cons (prim append head following : Bytes) (pairUp more rest))))),
  recursive_argument: fuel
}
