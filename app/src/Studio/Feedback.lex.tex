  name: formsOf,
  parameters: [lines : (List Bytes)],
  result: (Option (List Studio.Feedback.Intake::Form)),

    | List.nil => (ctor Option.some [(List Studio.Feedback.Intake::Form)] (nil Studio.Feedback.Intake::Form))
    | List.cons line rest => (match (Studio.Feedback.Intake::formOf line)
      | Option.none => (ctor Option.none [(List Studio.Feedback.Intake::Form)])

  recursive_argument: forms
