definition{
  name: planRecite,
  parameters: [seen : Studio.Ops::Seen, tag : Bytes],
  result: Studio.Ops::Plan,
  body: (let items : (List LexLeanStd.Json::Json) := (Studio.Ops::arrayOf seen (Studio.Doc::lit "model/admissions.json")); (let moved : (List LexLeanStd.Json::Json) := (reciteAll items (versionOf tag)); (if (Studio.Ops::sameList (citedOf items) (citedOf moved)) then (ctor Studio.Ops::Plan.done (nil (Bytes * Bytes)) (Studio.Doc::cat (cons (Studio.Doc::lit "verdant-engine ") (cons tag (cons (Studio.Doc::lit " is pinned, and every admission of its registers cites it\n") (nil Bytes)))))) else (ctor Studio.Ops::Plan.done (cons ((Studio.Doc::lit "model/admissions.json"), (LexLeanStd.Json::writeJsonFile (ctor LexLeanStd.Json::Json.JArr (Studio.Ops::jsonItems moved)))) (nil (Bytes * Bytes))) (Studio.Doc::cat (cons (Studio.Doc::lit "re-cited the admissions of the verdant-engine registers at ") (cons tag (cons 0x0a (nil Bytes))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: planFetch,
  parameters: [seen : Studio.Ops::Seen, tag : Bytes, commit : Bytes],
  result: Studio.Ops::Plan,
  body: (if ((prim length commit : Nat) =? 0) then (ctor Studio.Ops::Plan.done (nil (Bytes * Bytes)) (Studio.Doc::cat (cons (Studio.Doc::lit "waiting engine: the release tag ") (cons tag (cons (Studio.Doc::lit " names no commit\n") (nil Bytes)))))) else (match (firstUnfetched seen (registers))
    | Option.some register => (download (artifactOf register) (rawUrl register commit))
    | Option.none => (if (Studio.Ops::negate (Studio.Doc::memberBytes (datePath) (project seen ran))) then (ctor Studio.Ops::Plan.run (Studio.Doc::lit "date") (cons (Studio.Doc::lit "-u") (cons (Studio.Doc::cat (cons 0x2b2546 (nil Bytes))) (nil Bytes))) (datePath)) else (match (Studio.Ops::missing seen (cons (childOut) (artifactsOf (registers))))
      | List.cons head rest => (ctor Studio.Ops::Plan.reads (cons head rest))
      | List.nil => (ctor Studio.Ops::Plan.done (bindingWrites seen (registers) tag commit (LexLeanStd.Text::trim (Studio.Ops::contents seen (childOut)))) (Studio.Doc::cat (cons (Studio.Doc::lit "staged the verdant-engine ") (cons tag (cons (Studio.Doc::lit " bindings at ") (cons commit (cons 0x0a (nil Bytes)))))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: planPin,
  parameters: [seen : Studio.Ops::Seen, tag : Bytes],
  result: Studio.Ops::Plan,
  body: (if (Studio.Ops::negate (Studio.Doc::memberBytes (refPath) (project seen ran))) then (download (refPath) (api (Studio.Doc::cat (cons (Studio.Doc::lit "git/ref/tags/") (cons tag (nil Bytes)))))) else (match (Studio.Ops::missing seen (cons (refPath) (nil Bytes)))
    | List.cons head rest => (ctor Studio.Ops::Plan.reads (cons head rest))
    | List.nil => (let object : LexLeanStd.Json::Json := (match (Studio.Doc::field (match (Studio.Ops::jsonOf seen (refPath))
          | Option.none => (ctor LexLeanStd.Json::Json.JNil)
          | Option.some found => found) (Studio.Doc::lit "object"))
        | Option.none => (ctor LexLeanStd.Json::Json.JNil)
        | Option.some found => found); (if (Studio.Doc::same (Studio.Doc::str object (Studio.Doc::lit "type")) (Studio.Doc::lit "tag")) then (if (Studio.Ops::negate (Studio.Doc::memberBytes (tagPath) (project seen ran))) then (download (tagPath) (api (Studio.Doc::cat (cons (Studio.Doc::lit "git/tags/") (cons (Studio.Doc::str object (Studio.Doc::lit "sha")) (nil Bytes)))))) else (match (Studio.Ops::missing seen (cons (tagPath) (nil Bytes)))
          | List.cons head rest => (ctor Studio.Ops::Plan.reads (cons head rest))
          | List.nil => (planFetch seen tag (match (Studio.Doc::field (match (Studio.Ops::jsonOf seen (tagPath))
              | Option.none => (ctor LexLeanStd.Json::Json.JNil)
              | Option.some found => found) (Studio.Doc::lit "object"))
            | Option.none => 0x
            | Option.some found => (Studio.Doc::str found (Studio.Doc::lit "sha")))))) else (planFetch seen tag (Studio.Doc::str object (Studio.Doc::lit "sha"))))))),
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: planStageEngine,
  parameters: [seen : Studio.Ops::Seen],
  result: Studio.Ops::Plan,
  body: (if (Studio.Ops::negate (Studio.Doc::memberBytes (releasePath) (project seen ran))) then (download (releasePath) (api (Studio.Doc::lit "releases/latest"))) else (match (Studio.Ops::missing seen (cons (releasePath) (cons (Studio.Doc::lit "model/authorities.toml") (cons (Studio.Doc::lit "model/admissions.json") (nil Bytes)))))
    | List.cons head rest => (ctor Studio.Ops::Plan.reads (cons head rest))
    | List.nil => (let tag : Bytes := (match (Studio.Ops::jsonOf seen (releasePath))
        | Option.none => 0x
        | Option.some release => (Studio.Doc::str release (Studio.Doc::lit "tag_name"))); (if ((prim length tag : Nat) =? 0) then (ctor Studio.Ops::Plan.done (nil (Bytes * Bytes)) (Studio.Doc::lit "waiting engine: the latest verdant-engine release names no tag\n")) else (let entries : (List Bytes) := (idsAt (registers) (versionOf tag)); (if ((prim length (Studio.Ops::absent entries (Studio.Check::tomlIds (Studio.Ops::contents seen (Studio.Doc::lit "model/authorities.toml")))) : Nat) =? 0) then (planRecite seen tag) else (match (Studio.Ops::missing seen (bindingPaths entries))
      | List.cons head rest => (ctor Studio.Ops::Plan.reads (cons head rest))
      | List.nil => (if (Studio.Ops::present seen (bindingPath (idOf (ctor Register.ids) (versionOf tag)))) then (ctor Studio.Ops::Plan.done (nil (Bytes * Bytes)) (Studio.Doc::cat (cons (Studio.Doc::lit "waiting ingest: the verdant-engine ") (cons tag (cons (Studio.Doc::lit " bindings are staged, and advance ingests them\n") (nil Bytes)))))) else (planPin seen tag))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}
