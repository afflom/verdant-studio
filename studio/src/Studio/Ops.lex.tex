  parameters: [text : Bytes, offset : Nat, remaining : Nat, out : Bytes],
  result: Bytes,
  body: (match remaining
    | Nat.zero => out
    | Nat.succ rest => (match (prim index text offset : (Option UInt8))
      | Option.none => out

  recursive_argument: remaining,
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: queuedDecision,
  parameters: [seen : Seen, path : Bytes, recorded : (List LexLeanStd.Json::Json)],
  result: (Bool * Bytes),

(prim length times : Nat)) then (true, (Studio.Doc::cat (cons (Studio.Doc::lit "record-decision ") (cons path (cons (Studio.Doc::lit "\n") (nil Bytes)))))) else (false, (waiting path (Studio.Doc::lit "some of its decisions are recorded already, so it needs a corrected file"))))))),

definition{
  name: queuedAdmission,
  parameters: [seen : Seen, path : Bytes, lines : (List Bytes)],
  result: (Bool * Bytes),
  body: (let requirements : (List LexLeanStd.Json::Json) := (someItems (admitLines lines (Studio.Doc::lit "requirement"))); (let admissions : (List LexLeanStd.Json::Json) := (someItems (admitLines lines (Studio.Doc::lit "admission"))); (let names : (List Bytes) := (Studio.Check::concat (Studio.Check::idsOf requirements (Studio.Doc::lit "id")) (Studio.Check::idsOf admissions (Studio.Doc::lit "construct"))); (let held : (List Bytes) := (Studio.Check::idsOf (arrayOf seen (Studio.Doc::lit "model/requirements.json")) (Studio.Doc::lit "id")); (let fresh : (List Bytes) := (absent names (Studio.Check::concat held (Studio.Check::idsOf (arrayOf seen (Studio.Doc::lit "model/admissions.json")) (Studio.Doc::lit "construct")))); (if ((prim length fresh : Nat) =? 0) then (false, 0x) else (if (negate ((prim length fresh : Nat) =? (prim length names : Nat))) then (false, (waiting path (Studio.Doc::lit "some of its names are in the model already, so it needs a corrected file"))) else (let unresolved : (List Bytes) := (LexLeanStd.Order::sortedSet (Studio.Check::concat (absent (authoritiesCited admissions) (Studio.Check::tomlIds (contents seen (Studio.Doc::lit "model/authorities.toml")))) (absent (requirementsCited admissions) (Studio.Check::concat held (Studio.Check::idsOf requirements (Studio.Doc::lit "id")))))); (if ((prim length unresolved : Nat) =? 0) then (true, (Studio.Doc::cat (cons (Studio.Doc::lit "admit ") (cons path (cons (Studio.Doc::lit "\n") (nil Bytes)))))) else (false, (waiting path (Studio.Doc::cat (cons (Studio.Doc::lit "cites ") (cons (LexLeanStd.Text::joinWith unresolved (Studio.Doc::lit " ")) (cons (Studio.Doc::lit ", which are not in the model yet") (nil Bytes)))))))))))))))),
  axioms: [Quot.sound, propext],
  executable: true

      | List.cons head rest => (ctor Plan.lists (cons head rest))
      | List.nil => (match (mappingNeeds seen)
        | List.cons head rest => (ctor Plan.reads (cons head rest))
        | List.nil => (ctor Plan.done (nil (Bytes * Bytes)) (Studio.Doc::cat (cons (second found) (cons (Studio.Doc::cat (LexLeanStd.Order::sortedSet (unresolvedWaits (knownOf seen) (entriesOf seen (stagedPaths seen))))) (cons (inferenceWait seen) (nil Bytes))))))))))))),

      (if (Studio.Doc::same op (Studio.Doc::lit "build")) then (planBuild seen) else

  executable: true
