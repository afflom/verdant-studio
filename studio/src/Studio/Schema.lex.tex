      | Option.some name => (if (Studio.Doc::memberBytes name (fieldNamesOf escaped (keysOf rest))) then (ctor Result.error [Pieces, Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "two properties derive the field name ") (cons name (cons (Studio.Doc::lit " (an ASCII name is camel case, one that starts with a digit gets the prefix n, any other by its code points as uXXXX; when those names are not distinct, every name in the set keeps its letters and digits and writes each other character, a leading digit, u and U as a C11 universal character name, uXXXX or UXXXXXXXX)") (nil Bytes)))))) else (match (fieldType (second head) (prim append owner (pascal (first head)) : Bytes))
      | Result.error reason => (ctor Result.error [Pieces, Bytes] reason)
      | Result.ok found => (let typeText : Bytes := (if (Studio.Doc::memberBytes (first head) required) then (typedText found) else (Studio.Doc::cat (cons (Studio.Doc::lit "(Option ") (cons (typedText found) (cons (Studio.Doc::lit ")") (nil Bytes)))))); (match (fieldsWith escaped owner rest required positional)
        | Result.error reason => (ctor Result.error [Pieces, Bytes] reason)
        | Result.ok others => (ctor Result.ok [Pieces, Bytes] (ctor Pieces.mk (cons (if positional then typeText else (Studio.Doc::cat (cons (Studio.Doc::lit "    {name: ") (cons name (cons (Studio.Doc::lit ", type: ") (cons typeText (cons (Studio.Doc::lit "}") (nil Bytes)))))))) (piecesText others)) (appendJobs (typedJobs found) (piecesJobs others)))))))))),
  recursive_argument: properties,
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: enumName,
  parameters: [escaped : Bool, value : Value],
  result: (Option Bytes),
  body: (match value
    | Value.VStr text => (nameAs escaped text)
    | Value.VDec negative mantissa scale => (match (escapedName (prim append (Studio.Doc::lit "n") (decimalText false mantissa scale) : Bytes))
      | Option.none => (ctor Option.none [Bytes])
      | Option.some spelled => (ctor Option.some [Bytes] (if negative then (prim append (Studio.Doc::lit "nMinus") (LexLeanStd.Text::sliceOr spelled 1 (prim subtract (prim length spelled : Nat) 1 : Nat)) : Bytes) else spelled)))
    | Value.VFloat wide bits => (ctor Option.none [Bytes])
    | Value.VInt negative magnitude => (ctor Option.some [Bytes] (prim append (if (Studio.Doc::both negative (0 <? magnitude)) then (Studio.Doc::lit "nMinus") else (Studio.Doc::lit "n")) (LexLeanStd.Json::integerText false magnitude) : Bytes))
    | Value.VTag tagName tagged => (ctor Option.none [Bytes])
    | Value.VNull => (ctor Option.none [Bytes])
    | Value.VBool flag => (ctor Option.none [Bytes])
    | Value.VArr items => (ctor Option.none [Bytes])
    | Value.VObj members => (ctor Option.none [Bytes])
    | Value.VNil => (ctor Option.none [Bytes])
    | Value.VItem head rest => (ctor Option.none [Bytes])
    | Value.VMember key item rest => (ctor Option.none [Bytes])),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: constructorsWith,

      | Option.some name => (if (Studio.Doc::memberBytes name named) then (ctor Result.error [(List Bytes), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "two enum values derive the constructor name ") (cons name (cons (Studio.Doc::lit " (an ASCII string is named in camel case, any other string by its code points as uXXXX, an integer n is named nN, and a negative one nMinusN; when the string names are not distinct, every string keeps its letters and digits and writes each other character, a leading digit, u and U as a C11 universal character name, uXXXX or UXXXXXXXX)") (nil Bytes)))))) else (match (constructorsWith escaped rest (cons name named))

definition{
  name: branchesWith,
  parameters: [escaped : Bool, owner : Bytes, branches : (List Value), key : Bytes, named : (List Bytes)],
  result: (Result Pieces Bytes),
  body: (match branches
    | List.nil => (ctor Result.ok [Pieces, Bytes] (ctor Pieces.mk (nil Bytes) (nil Job)))
    | List.cons head rest => (match (tagIn head key)
      | Option.none => (ctor Result.error [Pieces, Bytes] (Studio.Doc::lit "a oneOf branch has no tag"))
      | Option.some tag => (match (nameAs escaped tag)
        | Option.none => (ctor Result.error [Pieces, Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "unnameable oneOf tag ") (cons tag (nil Bytes)))))
        | Option.some name => (if (Studio.Doc::memberBytes name named) then (ctor Result.error [Pieces, Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "two oneOf branches derive the constructor name ") (cons name (cons (Studio.Doc::lit ", so the oneOf is ambiguous") (nil Bytes)))))) else (match (fieldsOf (prim append owner (pascal tag) : Bytes) (withoutKey (membersOf (field head (Studio.Doc::lit "properties"))) key) (strings (itemsOf (field head (Studio.Doc::lit "required")))) true)
          | Result.error reason => (ctor Result.error [Pieces, Bytes] reason)
          | Result.ok found => (match (branchesWith escaped owner rest key (cons name named))
            | Result.error reason => (ctor Result.error [Pieces, Bytes] reason)
            | Result.ok others => (ctor Result.ok [Pieces, Bytes] (ctor Pieces.mk (cons (Studio.Doc::cat (cons (Studio.Doc::lit "    {name: ") (cons name (cons (Studio.Doc::lit ", fields: [") (cons (joinWith (piecesText found) (Studio.Doc::lit ", ")) (cons (Studio.Doc::lit "]}") (nil Bytes))))))) (piecesText others)) (appendJobs (piecesJobs found) (piecesJobs others)))))))))),
  recursive_argument: branches,
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: unionWith,
  parameters: [escaped : Bool, owner : Bytes, branches : (List Value), named : (List Bytes)],
  result: (Result Pieces Bytes),
  body: (match branches
    | List.nil => (ctor Result.ok [Pieces, Bytes] (ctor Pieces.mk (nil Bytes) (nil Job)))
    | List.cons head rest => (match (branchLabel head)
      | Option.none => (ctor Result.error [Pieces, Bytes] (Studio.Doc::lit "untagged oneOf: a branch has neither a $ref nor a title, so its constructor has no name"))
      | Option.some label => (match (nameAs escaped label)
        | Option.none => (ctor Result.error [Pieces, Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "unnameable oneOf branch ") (cons label (nil Bytes)))))
        | Option.some name => (if (Studio.Doc::memberBytes name named) then (ctor Result.error [Pieces, Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "two oneOf branches derive the constructor name ") (cons name (cons (Studio.Doc::lit ", so the oneOf is ambiguous") (nil Bytes)))))) else (match (baseType head (prim append owner (pascal label) : Bytes))
          | Result.error reason => (ctor Result.error [Pieces, Bytes] reason)
          | Result.ok found => (match (unionWith escaped owner rest (cons name named))
            | Result.error reason => (ctor Result.error [Pieces, Bytes] reason)
            | Result.ok others => (ctor Result.ok [Pieces, Bytes] (ctor Pieces.mk (cons (Studio.Doc::cat (cons (Studio.Doc::lit "    {name: ") (cons name (cons (Studio.Doc::lit ", fields: [") (cons (typedText found) (cons (Studio.Doc::lit "]}") (nil Bytes))))))) (piecesText others)) (appendJobs (typedJobs found) (piecesJobs others)))))))))),
  recursive_argument: branches,
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: taggedDeclaration,

      | List.nil => (match (unionWith (collides (labelsOf branches)) name branches (nil Bytes))

  executable: true
}

    | List.cons property others => (match (fieldsOf name (membersOf (field schema (Studio.Doc::lit "properties"))) (strings (itemsOf (field schema (Studio.Doc::lit "required")))) false)
      | Result.error reason => (ctor Result.error [Typed, Bytes] reason)
      | Result.ok found => (typed (Studio.Doc::cat (cons (Studio.Doc::lit "structure{\n  name: ") (cons name (cons (Studio.Doc::lit ",\n  type_parameters: [],\n  parameters: [],\n  fields: [") (cons (match (piecesText found)
        | List.nil => 0x
        | List.cons head rest => (Studio.Doc::cat (cons (Studio.Doc::lit "\n") (cons (joinWith (piecesText found) (Studio.Doc::lit ",\n")) (cons (Studio.Doc::lit "\n  ") (nil Bytes)))))) (cons (Studio.Doc::lit "]\n}\n") (nil Bytes))))))) (piecesJobs found))))))),
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: expand,
  parameters: [fuel : Nat, jobs : (List Job), done : (List Bytes), out : (List Bytes)],
  result: (Result (List Bytes) Bytes),
  body: (match fuel
    | Nat.zero => (ctor Result.error [(List Bytes), Bytes] (Studio.Doc::lit "schema nests deeper than the translator's bound"))
    | Nat.succ rest => (match jobs
      | List.nil => (ctor Result.ok [(List Bytes), Bytes] out)
      | List.cons head tail => (match head
        | Job.emit text => (expand rest tail done (cons text out))
        | Job.json => (if (Studio.Doc::memberBytes (jsonValueName) done) then (expand rest tail done out) else (expand rest tail (cons (jsonValueName) done) (cons (jsonValueDeclaration) out)))
        | Job.translate name schema => (if (Studio.Doc::same name (jsonValueName)) then (ctor Result.error [(List Bytes), Bytes] (Studio.Doc::lit "the type name JsonValue is reserved for schemas that admit any value")) else (if (Studio.Doc::memberBytes name done) then (expand rest tail done out) else (match (declaration name schema)
          | Result.error reason => (ctor Result.error [(List Bytes), Bytes] reason)
          | Result.ok found => (expand rest (appendJobs (typedJobs found) (cons (ctor Job.emit (typedText found)) tail)) (cons name done) out))))))),
  recursive_argument: fuel,
  axioms: [Quot.sound, propext],
  executable: true
}

    | Value.VObj members => (refsInto members found)
    | Value.VItem head rest => (refsInto head (refsInto rest found))
    | Value.VMember key item rest => (let later : (List Bytes) := (refsInto item (refsInto rest found)); (if (Studio.Doc::same key (Studio.Doc::lit "$ref")) then (match (strOf (ctor Option.some [Value] item))
      | Option.none => later
      | Option.some reference => (match (refName reference)
        | Option.none => later
        | Option.some named => (cons named later))) else later))
    | Value.VTag tagName tagged => found
    | Value.VNull => found
    | Value.VBool flag => found
    | Value.VDec negative mantissa scale => found

    | Value.VInt negative magnitude => found

definition{
  name: orderedDefs,
  parameters: [fuel : Nat, pending : (List (Bytes * Value)), placed : (List Bytes), names : (List Bytes)],
  result: (Result (List (Bytes * Value)) Bytes),
  body: (match fuel
    | Nat.zero => (match pending
      | List.nil => (ctor Result.ok [(List (Bytes * Value)), Bytes] (nil (Bytes * Value)))
      | List.cons any others => (ctor Result.error [(List (Bytes * Value)), Bytes] (Studio.Doc::lit "the definitions are more than the translator's bound")))
    | Nat.succ rest => (match pending
      | List.nil => (ctor Result.ok [(List (Bytes * Value)), Bytes] (nil (Bytes * Value)))
      | List.cons any others => (match (firstReady pending placed names)
        | Option.none => (ctor Result.error [(List (Bytes * Value)), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "these definitions refer to each other in a cycle, which no declaration order can satisfy: ") (cons (joinWith (defNames pending) (Studio.Doc::lit ", ")) (nil Bytes)))))
        | Option.some next => (match (orderedDefs rest (withoutKey pending (first next)) (cons (typeName (first next)) placed) names)
          | Result.error reason => (ctor Result.error [(List (Bytes * Value)), Bytes] reason)
          | Result.ok later => (ctor Result.ok [(List (Bytes * Value)), Bytes] (cons next later)))))),
  recursive_argument: fuel,
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: translateSchema,
  parameters: [schema : Value, module : Bytes, root : Bytes],
  result: (Result Bytes Bytes),
  body: (let definitions : (List (Bytes * Value)) := (appendMembers (membersOf (field schema (Studio.Doc::lit "$defs"))) (membersOf (field schema (Studio.Doc::lit "definitions")))); (match (orderedDefs 4096 definitions (nil Bytes) (defNames definitions))
    | Result.error reason => (ctor Result.error [Bytes, Bytes] reason)
    | Result.ok ordered => (match (expand 4096 (appendJobs (defJobs ordered) (cons (ctor Job.translate root schema) (nil Job))) (nil Bytes) (nil Bytes))
    | Result.error reason => (ctor Result.error [Bytes, Bytes] reason)
    | Result.ok declarations => (ctor Result.ok [Bytes, Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "\\begin{lexlean}{") (cons module (cons (Studio.Doc::lit "}\n\\useglossary{lexlean.std.nat@1.2.0}\n\\title{Natural number addition}\n\\begin{semanticmodule}\n") (cons (joinWith (reverseBytes declarations (nil Bytes)) (Studio.Doc::lit "\n")) (cons (Studio.Doc::lit "\\end{semanticmodule}\n\\end{lexlean}\n") (nil Bytes))))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}
