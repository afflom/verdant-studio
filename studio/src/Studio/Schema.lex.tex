  parameters: [text : Bytes, offset : Nat, remaining : Nat, raise : Bool, out : Bytes],
  result: Bytes,
  body: (match remaining
    | Nat.zero => out
    | Nat.succ rest => (match (prim index text offset : (Option UInt8))
      | Option.none => out
      | Option.some byte => (let value : Nat := (prim to_nat byte : Nat); (if (isSeparator value) then (caseFrom text (offset + 1) rest true out) else (if ((lowerLetter value) && raise) then (caseFrom text (offset + 1) rest false (prim append out (LexLeanStd.Text::sliceOr (prim utf8_encode "ABCDEFGHIJKLMNOPQRSTUVWXYZ" : Bytes) (prim subtract value 97 : Nat) 1) : Bytes)) else (if ((lowerLetter value) || ((upperLetter value) || (digit value))) then (caseFrom text (offset + 1) rest false (prim append out (LexLeanStd.Text::sliceOr text offset 1) : Bytes)) else (caseFrom text (offset + 1) rest raise out))))))),

  axioms: [Quot.sound, propext],

    | LexLeanStd.Json::Json.JInt negative magnitude => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JStr text => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JArr items => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)

    | LexLeanStd.Json::Json.JMember key item rest => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JObj members => (match (Studio.Doc::strOf (Studio.Doc::field extra (Studio.Doc::lit "$ref")))

definition{
  name: baseType,
  parameters: [schema : LexLeanStd.Json::Json, context : Bytes],
  result: (Result (Bytes * (List Job)) Bytes),
  body: (match schema
    | LexLeanStd.Json::Json.JBool accepts => (if accepts then (anyValue) else (refused (Studio.Doc::lit "the false schema admits no instance") context))
    | LexLeanStd.Json::Json.JInt negative magnitude => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JStr text => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JArr items => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JNil => (anyValue)
    | LexLeanStd.Json::Json.JItem head rest => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JMember key item rest => (refused (Studio.Doc::lit "a schema must be an object or a boolean") context)
    | LexLeanStd.Json::Json.JObj members => (match (unknownKeyword (Studio.Doc::memberSpine schema))
    | Option.some keyword => (refused (Studio.Doc::cat (cons (Studio.Doc::lit "unsupported keyword ") (cons keyword (nil Bytes)))) context)
    | Option.none => (match (Studio.Doc::strOf (Studio.Doc::field schema (Studio.Doc::lit "$ref")))
      | Option.some reference => (match (refName reference)
        | Option.none => (refused (Studio.Doc::lit "unsupported reference") context)
        | Option.some named => (typed named (nil Job)))
      | Option.none => (match (Studio.Doc::field schema (Studio.Doc::lit "allOf"))
      | Option.some branches => (match (Studio.Doc::spineItems branches)
        | List.nil => (refused (Studio.Doc::lit "an empty allOf has no schema to translate") context)
        | List.cons only others => (match others
          | List.cons second1 more => (refused (Studio.Doc::lit "an allOf with more than one branch is not translated, because no standard defines how its branches merge into one type") context)
          | List.nil => (match (Studio.Doc::strOf (Studio.Doc::field only (Studio.Doc::lit "$ref")))
            | Option.none => (refused (Studio.Doc::lit "an allOf is translated only when its one branch is a $ref") context)
            | Option.some reference => (match (refName reference)
              | Option.none => (refused (Studio.Doc::lit "unsupported reference") context)
              | Option.some named => (typed named (nil Job))))))
      | Option.none => (match (Studio.Doc::field schema (Studio.Doc::lit "enum"))
        | Option.some values => (typed context (cons (ctor Job.translate context schema) (nil Job)))
        | Option.none => (match (Studio.Doc::field schema (Studio.Doc::lit "const"))
          | Option.some constant => (match (Studio.Doc::strOf (ctor Option.some [LexLeanStd.Json::Json] constant))
            | Option.some text => (typed (Studio.Doc::lit "Bytes") (nil Job))
            | Option.none => (refused (Studio.Doc::lit "unsupported const") context))
          | Option.none => (match (Studio.Doc::field schema (Studio.Doc::lit "oneOf"))
            | Option.some branches => (typed context (cons (ctor Job.translate context schema) (nil Job)))
            | Option.none => (let kind : Bytes := (Studio.Doc::str schema (Studio.Doc::lit "type")); (if ((Studio.Doc::same kind 0x) && (if (structural schema) then false else true)) then (anyValue) else (if (Studio.Doc::same kind (Studio.Doc::lit "string")) then (typed (Studio.Doc::lit "Bytes") (nil Job)) else (if (Studio.Doc::same kind (Studio.Doc::lit "integer")) then (typed (integerType schema) (nil Job)) else (if (Studio.Doc::same kind (Studio.Doc::lit "number")) then (typed (numberType) (nil Job)) else (if (Studio.Doc::same kind (Studio.Doc::lit "boolean")) then (typed (Studio.Doc::lit "Bool") (nil Job)) else (if (Studio.Doc::same kind (Studio.Doc::lit "object")) then (match (Studio.Doc::membersOf (Studio.Doc::field schema (Studio.Doc::lit "properties")))
              | List.cons property others => (typed context (cons (ctor Job.translate context schema) (nil Job)))
              | List.nil => (match (Studio.Doc::field schema (Studio.Doc::lit "additionalProperties"))
                | Option.none => (wrapped (Studio.Doc::lit "List") (pairedWith (anyValue)))
                | Option.some extra => (wrapped (Studio.Doc::lit "List") (pairedWith (mapValue extra context))))) else (refused (Studio.Doc::cat (cons (Studio.Doc::lit "unsupported type ") (cons kind (nil Bytes)))) context))))))))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: fieldsWith,
  parameters: [escaped : Bool, owner : Bytes, properties : (List (Bytes * LexLeanStd.Json::Json)), required : (List Bytes), positional : Bool],
  result: (Result ((List Bytes) * (List Job)) Bytes),
  body: (match properties
    | List.nil => (ctor Result.ok [((List Bytes) * (List Job)), Bytes] ((nil Bytes), (nil Job)))
    | List.cons head rest => (match (nameAs escaped (first head))
      | Option.none => (ctor Result.error [((List Bytes) * (List Job)), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "unnameable property ") (cons (first head) (nil Bytes)))))
      | Option.some name => (if (Studio.Doc::memberBytes name (namesOf escaped (keysOf rest))) then (ctor Result.error [((List Bytes) * (List Job)), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "two properties derive the field name ") (cons name (cons (Studio.Doc::lit " (an ASCII name is camel case, one that starts with a digit gets the prefix n, any other by its code points as uXXXX; when those names are not distinct, every name in the set keeps its letters and digits and writes each other character, a leading digit, u and U as a C11 universal character name, uXXXX or UXXXXXXXX)") (nil Bytes)))))) else (match (fieldType (second head) (prim append owner (pascal (first head)) : Bytes))
      | Result.error reason => (ctor Result.error [((List Bytes) * (List Job)), Bytes] reason)
      | Result.ok found => (let typeText : Bytes := (if (Studio.Doc::memberBytes (first head) required) then (first found) else (Studio.Doc::cat (cons (Studio.Doc::lit "(Option ") (cons (first found) (cons (Studio.Doc::lit ")") (nil Bytes)))))); (match (fieldsWith escaped owner rest required positional)
        | Result.error reason => (ctor Result.error [((List Bytes) * (List Job)), Bytes] reason)
        | Result.ok others => (ctor Result.ok [((List Bytes) * (List Job)), Bytes] ((cons (if positional then typeText else (Studio.Doc::cat (cons (Studio.Doc::lit "    {name: ") (cons name (cons (Studio.Doc::lit ", type: ") (cons typeText (cons (Studio.Doc::lit "}") (nil Bytes)))))))) (first others)), (appendJobs (second found) (second others)))))))))),
  recursive_argument: properties,
  axioms: [Quot.sound, propext],
  executable: true
}

definition{

definition{
  name: constructorsWith,

  name: branchesWith,
  parameters: [escaped : Bool, owner : Bytes, branches : (List LexLeanStd.Json::Json), key : Bytes, named : (List Bytes)],
  result: (Result ((List Bytes) * (List Job)) Bytes),
  body: (match branches
    | List.nil => (ctor Result.ok [((List Bytes) * (List Job)), Bytes] ((nil Bytes), (nil Job)))
    | List.cons head rest => (match (tagIn head key)
      | Option.none => (ctor Result.error [((List Bytes) * (List Job)), Bytes] (Studio.Doc::lit "a oneOf branch has no tag"))
      | Option.some tag => (match (nameAs escaped tag)
        | Option.none => (ctor Result.error [((List Bytes) * (List Job)), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "unnameable oneOf tag ") (cons tag (nil Bytes)))))
        | Option.some name => (if (Studio.Doc::memberBytes name named) then (ctor Result.error [((List Bytes) * (List Job)), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "two oneOf branches derive the constructor name ") (cons name (cons (Studio.Doc::lit ", so the oneOf is ambiguous") (nil Bytes)))))) else (match (fieldsOf (prim append owner (pascal tag) : Bytes) (withoutKey (Studio.Doc::membersOf (Studio.Doc::field head (Studio.Doc::lit "properties"))) key) (Studio.Doc::strings (Studio.Doc::itemsOf (Studio.Doc::field head (Studio.Doc::lit "required")))) true)
          | Result.error reason => (ctor Result.error [((List Bytes) * (List Job)), Bytes] reason)
          | Result.ok found => (match (branchesWith escaped owner rest key (cons name named))
            | Result.error reason => (ctor Result.error [((List Bytes) * (List Job)), Bytes] reason)
            | Result.ok others => (ctor Result.ok [((List Bytes) * (List Job)), Bytes] ((cons (Studio.Doc::cat (cons (Studio.Doc::lit "    {name: ") (cons name (cons (Studio.Doc::lit ", fields: [") (cons (joinWith (first found) (Studio.Doc::lit ", ")) (cons (Studio.Doc::lit "]}") (nil Bytes))))))) (first others)), (appendJobs (second found) (second others)))))))))),

  executable: true

definition{
  name: taggedDeclaration,
  parameters: [name : Bytes, branches : (List LexLeanStd.Json::Json)],
  result: (Result (Bytes * (List Job)) Bytes),
  body: (match branches
    | List.nil => (refused (Studio.Doc::lit "an empty oneOf admits no instance") name)
    | List.cons first1 others1 => (match (discriminators (Studio.Doc::membersOf (Studio.Doc::field first1 (Studio.Doc::lit "properties"))) branches)
      | List.nil => (refused (Studio.Doc::lit "untagged oneOf: no property is required by every branch with a const string") name)
      | List.cons key more => (match more
        | List.cons extra rest => (refused (Studio.Doc::cat (cons (Studio.Doc::lit "ambiguous oneOf: more than one property tags every branch (") (cons key (cons (Studio.Doc::lit ", ") (cons extra (cons (Studio.Doc::lit ")") (nil Bytes))))))) name)
        | List.nil => (match (branchConstructors name branches key (nil Bytes))
          | Result.error reason => (refused reason name)
          | Result.ok found => (typed (inductiveText name (first found)) (second found)))))),

definition{
  name: declaration,
  parameters: [name : Bytes, schema : LexLeanStd.Json::Json],
  result: (Result (Bytes * (List Job)) Bytes),
  body: (match (Studio.Doc::field schema (Studio.Doc::lit "enum"))
    | Option.some values => (match (constructorsOf (Studio.Doc::spineItems values) (nil Bytes))
      | Result.error reason => (refused reason name)
      | Result.ok constructors => (typed (Studio.Doc::cat (cons (Studio.Doc::lit "inductive{\n  name: ") (cons name (cons (Studio.Doc::lit ",\n  type_parameters: [],\n  parameters: [],\n  constructors: [\n") (cons (joinWith constructors (Studio.Doc::lit ",\n")) (cons (Studio.Doc::lit "\n  ]\n}\n") (nil Bytes))))))) (nil Job)))
    | Option.none => (match (Studio.Doc::field schema (Studio.Doc::lit "oneOf"))
    | Option.some branches => (taggedDeclaration name (Studio.Doc::spineItems branches))
    | Option.none => (if (refersTo schema name) then (match (fieldsOf name (Studio.Doc::membersOf (Studio.Doc::field schema (Studio.Doc::lit "properties"))) (Studio.Doc::strings (Studio.Doc::itemsOf (Studio.Doc::field schema (Studio.Doc::lit "required")))) true)
      | Result.error reason => (ctor Result.error [(Bytes * (List Job)), Bytes] reason)
      | Result.ok found => (typed (inductiveText name (cons (Studio.Doc::cat (cons (Studio.Doc::lit "    {name: mk, fields: [") (cons (joinWith (first found) (Studio.Doc::lit ", ")) (cons (Studio.Doc::lit "]}") (nil Bytes))))) (nil Bytes))) (second found))) else (match (Studio.Doc::membersOf (Studio.Doc::field schema (Studio.Doc::lit "properties")))
    | List.nil => (match (fieldType schema (prim append name (Studio.Doc::lit "Value") : Bytes))
      | Result.error reason => (ctor Result.error [(Bytes * (List Job)), Bytes] reason)
      | Result.ok found => (typed (Studio.Doc::cat (cons (Studio.Doc::lit "structure{\n  name: ") (cons name (cons (Studio.Doc::lit ",\n  type_parameters: [],\n  parameters: [],\n  fields: [\n    {name: value, type: ") (cons (first found) (cons (Studio.Doc::lit "}\n  ]\n}\n") (nil Bytes))))))) (second found)))
    | List.cons property others => (match (fieldsOf name (Studio.Doc::membersOf (Studio.Doc::field schema (Studio.Doc::lit "properties"))) (Studio.Doc::strings (Studio.Doc::itemsOf (Studio.Doc::field schema (Studio.Doc::lit "required")))) false)
      | Result.error reason => (ctor Result.error [(Bytes * (List Job)), Bytes] reason)
      | Result.ok found => (typed (Studio.Doc::cat (cons (Studio.Doc::lit "structure{\n  name: ") (cons name (cons (Studio.Doc::lit ",\n  type_parameters: [],\n  parameters: [],\n  fields: [") (cons (match (first found)
        | List.nil => 0x
        | List.cons head rest => (Studio.Doc::cat (cons (Studio.Doc::lit "\n") (cons (joinWith (first found) (Studio.Doc::lit ",\n")) (cons (Studio.Doc::lit "\n  ") (nil Bytes)))))) (cons (Studio.Doc::lit "]\n}\n") (nil Bytes))))))) (second found))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: expand,
  parameters: [fuel : Nat, jobs : (List Job), done : (List Bytes), out : (List Bytes)],
  result: (Result (List Bytes) Bytes),
  body: (match fuel
    | Nat.zero => (ctor Result.error [(List Bytes), Bytes] (Studio.Doc::lit "schema nests deeper than the translator's bound"))
    | Nat.succ rest => (match jobs
      | List.nil => (ctor Result.ok [(List Bytes), Bytes] out)
      | List.cons head tail => (match head
        | Job.emit text => (expand rest tail done (cons text out))
        | Job.json => (if (Studio.Doc::memberBytes (jsonValueName) done) then (expand rest tail done out) else (expand rest tail (cons (jsonValueName) done) (cons (jsonValueDeclaration) out)))
        | Job.translate name schema => (if (Studio.Doc::same name (jsonValueName)) then (ctor Result.error [(List Bytes), Bytes] (Studio.Doc::lit "the type name JsonValue is reserved for schemas that admit any value")) else (if (Studio.Doc::memberBytes name done) then (expand rest tail done out) else (match (declaration name schema)
          | Result.error reason => (ctor Result.error [(List Bytes), Bytes] reason)
          | Result.ok found => (expand rest (appendJobs (second found) (cons (ctor Job.emit (first found)) tail)) (cons name done) out))))))),
  recursive_argument: fuel,
  axioms: [Quot.sound, propext],
  executable: true
}
