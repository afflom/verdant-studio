definition{
  name: mulAdd,
  parameters: [digits : (List Nat), factor : Nat, carry : Nat],
  result: (List Nat),
  body: (match digits
    | List.nil => (if (carry =? 0) then (nil Nat) else (cons carry (nil Nat)))
    | List.cons head rest => (let value : Nat := ((prim multiply head factor : Nat) + carry); (cons (prim remainder value (limb) 0 : Nat) (mulAdd rest factor (prim quotient value (limb) 0 : Nat))))),
  recursive_argument: digits,
  executable: true
}

definition{
  name: fromNat,
  parameters: [fuel : Nat, number : Nat],
  result: (List Nat),
  body: (match fuel
    | Nat.zero => (nil Nat)
    | Nat.succ rest => (if (number =? 0) then (nil Nat) else (cons (prim remainder number (limb) 0 : Nat) (fromNat rest (prim quotient number (limb) 0 : Nat))))),
  recursive_argument: fuel,
  executable: true
}

definition{
  name: repeatTimes,
  parameters: [count : Nat, digits : (List Nat), factor : Nat],
  result: (List Nat),
  body: (match count
    | Nat.zero => digits
    | Nat.succ rest => (repeatTimes rest (mulAdd digits factor 0) factor)),
  recursive_argument: count,
  executable: true
}

definition{
  name: limbCount,
  parameters: [digits : (List Nat)],
  result: Nat,
  body: (match digits
    | List.nil => 0
    | List.cons head rest => ((limbCount rest) + 1)),
  recursive_argument: digits,
  executable: true
}

definition{
  name: compareSame,
  parameters: [left : (List Nat), right : (List Nat)],
  result: Nat,
  body: (match left
    | List.nil => 1
    | List.cons head rest => (match right
      | List.nil => 2
      | List.cons other later => (let high : Nat := (compareSame rest later); (if (high =? 1) then (if (head <? other) then 0 else (if (other <? head) then 2 else 1)) else high)))),
  recursive_argument: left,
  executable: true
}

definition{
  name: compare,
  parameters: [left : (List Nat), right : (List Nat)],
  result: Nat,
  body: (let a : Nat := (limbCount left); (let b : Nat := (limbCount right); (if (a <? b) then 0 else (if (b <? a) then 2 else (compareSame left right))))),
  executable: true
}

definition{
  name: trimHigh,
  parameters: [digits : (List Nat)],
  result: (List Nat),
  body: (match digits
    | List.nil => (nil Nat)
    | List.cons head rest => (let high : (List Nat) := (trimHigh rest); (match high
      | List.nil => (if (head =? 0) then (nil Nat) else (cons head (nil Nat)))
      | List.cons top more => (cons head high)))),
  recursive_argument: digits,
  executable: true
}

definition{
  name: subtractFrom,
  parameters: [left : (List Nat), right : (List Nat), borrow : Nat],
  result: (List Nat),
  body: (match left
    | List.nil => (nil Nat)
    | List.cons head rest => (let need : Nat := ((match right
      | List.nil => 0
      | List.cons other later => other) + borrow); (let tail : (List Nat) := (match right
      | List.nil => (nil Nat)
      | List.cons other later => later); (if (need <=? head) then (cons (prim subtract head need : Nat) (subtractFrom rest tail 0)) else (cons (prim subtract (head + (limb)) need : Nat) (subtractFrom rest tail 1)))))),
  recursive_argument: left,
  executable: true
}

definition{
  name: minus,
  parameters: [left : (List Nat), right : (List Nat)],
  result: (List Nat),
  body: (trimHigh (subtractFrom left right 0)),
  executable: true
}

definition{
  name: bitsFrom,
  parameters: [fuel : Nat, probe : (List Nat), digits : (List Nat), count : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => count
    | Nat.succ rest => (if ((compare probe digits) =? 2) then count else (bitsFrom rest (mulAdd probe 2 0) digits (count + 1)))),
  recursive_argument: fuel,
  executable: true
}

definition{
  name: bitLength,
  parameters: [fuel : Nat, digits : (List Nat)],

  executable: true
}

definition{

  executable: true
