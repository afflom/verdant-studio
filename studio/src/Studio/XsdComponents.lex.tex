definition{
  name: nodes,
  parameters: [content : Studio.Xml::Xml, scope : (List (Bytes * Bytes))],
  result: (Result (List Node) Bytes),
  body: (match content
    | Studio.Xml::Xml.XItem head rest => (match (nodes rest scope)
      | Result.error reason => (ctor Result.error [(List Node), Bytes] reason)
      | Result.ok later => (match head
        | Studio.Xml::Xml.XElem label attrs inner => (match (Studio.Xsd::declarations attrs scope)
          | Result.error reason => (ctor Result.error [(List Node), Bytes] reason)
          | Result.ok local => (match (Studio.Xsd::expand local label)
            | Result.error reason => (ctor Result.error [(List Node), Bytes] reason)
            | Result.ok expanded => (if ((Studio.Doc::same (first expanded) (Studio.Xsd::xsdNamespace)) && (negate (Studio.Doc::same (second expanded) (Studio.Doc::lit "annotation")))) then (ctor Result.ok [(List Node), Bytes] (cons (record Node {space := (first expanded), kind := (second expanded), attrs := attrs, content := inner, scope := local}) later)) else (ctor Result.ok [(List Node), Bytes] later))))
        | Studio.Xml::Xml.XAttr label value more => (ctor Result.ok [(List Node), Bytes] later)
        | Studio.Xml::Xml.XText text => (ctor Result.ok [(List Node), Bytes] later)
        | Studio.Xml::Xml.XItem first1 more => (ctor Result.ok [(List Node), Bytes] later)
        | Studio.Xml::Xml.XNil => (ctor Result.ok [(List Node), Bytes] later)))
    | Studio.Xml::Xml.XElem label attrs inner => (ctor Result.ok [(List Node), Bytes] (nil Node))
    | Studio.Xml::Xml.XAttr label value more => (ctor Result.ok [(List Node), Bytes] (nil Node))
    | Studio.Xml::Xml.XText text => (ctor Result.ok [(List Node), Bytes] (nil Node))
    | Studio.Xml::Xml.XNil => (ctor Result.ok [(List Node), Bytes] (nil Node))),
  recursive_argument: content,
  executable: true
}

definition{
  name: typeRef,
  parameters: [context : Context, node : Node, qname : Bytes],
  result: (Result (Bytes * (List Bytes)) Bytes),
  body: (match (Studio.Xsd::expand (project node scope) qname)
    | Result.error reason => (ctor Result.error [(Bytes * (List Bytes)), Bytes] reason)
    | Result.ok expanded => (if (Studio.Doc::same (first expanded) (Studio.Xsd::xsdNamespace)) then (if (Studio.Doc::same (second expanded) (Studio.Doc::lit "anyType")) then (ctor Result.ok [(Bytes * (List Bytes)), Bytes] ((xmlNodeName), (cons (xmlNodeName) (nil Bytes)))) else (match (Studio.Xsd::builtinType (second expanded) 16)
      | Result.error reason => (ctor Result.error [(Bytes * (List Bytes)), Bytes] reason)
      | Result.ok found => (ctor Result.ok [(Bytes * (List Bytes)), Bytes] (found, (nil Bytes))))) else (if (Studio.Doc::memberBytes (first expanded) (project context targets)) then (let named : Bytes := (Studio.Schema::pascal (second expanded)); (match (declared context "complexType" expanded)
      | Option.some found => (ctor Result.ok [(Bytes * (List Bytes)), Bytes] (named, (cons named (nil Bytes))))
      | Option.none => (match (declared context "simpleType" expanded)
        | Option.some found => (ctor Result.ok [(Bytes * (List Bytes)), Bytes] (named, (cons named (nil Bytes))))
        | Option.none => (ctor Result.error [(Bytes * (List Bytes)), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "the type ") (cons qname (cons (Studio.Doc::lit " is not defined by any schema document of this set (XSD 1.1 Part 1, section 3.17.6.2)") (nil Bytes))))))))) else (ctor Result.error [(Bytes * (List Bytes)), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "the type ") (cons qname (cons (Studio.Doc::lit " is in the namespace ") (cons (first expanded) (cons (Studio.Doc::lit ", which no schema document of this set targets (XSD 1.1 Part 1, sections 3.17.6.2 and 4.2.6)") (nil Bytes))))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: namesWith,
  parameters: [escaped : Bool, labels : (List Bytes), named : (List Bytes)],
  result: (Result (List Bytes) Bytes),
  body: (match labels
    | List.nil => (ctor Result.ok [(List Bytes), Bytes] (nil Bytes))
    | List.cons head rest => (match (Studio.Schema::nameAs escaped head)
      | Option.none => (ctor Result.error [(List Bytes), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "the component name ") (cons head (cons (Studio.Doc::lit " derives no LexLean name") (nil Bytes))))))
      | Option.some name => (if (Studio.Doc::memberBytes name named) then (ctor Result.error [(List Bytes), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "two components derive the name ") (cons name (cons (Studio.Doc::lit " (an ASCII name is camel case, one that starts with a digit gets the prefix n; when those names are not distinct, every name keeps its letters and digits and writes each other character as a C11 universal character name)") (nil Bytes)))))) else (match (namesWith escaped rest (cons name named))
        | Result.error reason => (ctor Result.error [(List Bytes), Bytes] reason)
        | Result.ok others => (ctor Result.ok [(List Bytes), Bytes] (cons name others)))))),
  recursive_argument: labels,
  axioms: [Quot.sound, propext],
  executable: true
}

  body: (match (attr node "type")
    | Option.some qname => (match (typeRef context node qname)
      | Result.error reason => (ctor Result.error [((Bytes * (List Bytes)) * (List Decl)), Bytes] (located reason name))
      | Result.ok found => (ctor Result.ok [((Bytes * (List Bytes)) * (List Decl)), Bytes] (found, (nil Decl))))
    | Option.none => (match (children node)
      | Result.error reason => (ctor Result.error [((Bytes * (List Bytes)) * (List Decl)), Bytes] reason)
      | Result.ok inner => (match (firstOf inner (cons (Studio.Doc::lit "simpleType") (nil Bytes)))
        | Option.none => (ctor Result.error [((Bytes * (List Bytes)) * (List Decl)), Bytes] (located (Studio.Doc::lit "an attribute declaration with no type has the type xs:anySimpleType, which is not translated, because it is the union of every simple value space (XSD 1.1 Part 1, section 3.2.2.1; Part 2, section 3.2.1)") name))
        | Option.some simple => (match (simpleDecl name simple space)
          | Result.error reason => (ctor Result.error [((Bytes * (List Bytes)) * (List Decl)), Bytes] reason)
          | Result.ok decl => (ctor Result.ok [((Bytes * (List Bytes)) * (List Decl)), Bytes] ((name, (cons name (nil Bytes))), (cons decl (nil Decl)))))))),
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: attributes,
  parameters: [fuel : Nat, context : Context, owner : Bytes, space : Bytes, items : (List Node), acc : Gathered],
  result: (Result Gathered Bytes),
  body: (match fuel
    | Nat.zero => (ctor Result.error [Gathered, Bytes] (located (Studio.Doc::lit "the attribute group references nest deeper than the schema set has components, so they refer to each other in a cycle (XSD 1.1 Part 1, section 3.6.3)") owner))
    | Nat.succ rest => (match items
      | List.nil => (ctor Result.ok [Gathered, Bytes] acc)
      | List.cons node more => (if (isKind node "attribute") then (if (attrIs node "use" "prohibited") then (attributes rest context owner space more acc) else (let required : Bool := (attrIs node "use" "required"); (match (attr node "ref")

      | Result.ok later => (if ((isKind head "redefine") || ((isKind head "override") || (isKind head "defaultOpenContent"))) then (ctor Result.error [(List (Bytes * (Bytes * Bytes))), Bytes] (Studio.Doc::cat (cons (project doc path) (cons (Studio.Doc::lit ": xs:") (cons (project head kind) (cons (Studio.Doc::lit " is not translated, because it changes components that other schema documents define (XSD 1.1 Part 1, sections 4.2.4, 4.2.5 and 3.4.2.3)") (nil Bytes))))))) else (if ((isKind head "import") || (isKind head "include")) then (match (attr head "schemaLocation")
        | Option.none => (ctor Result.error [(List (Bytes * (Bytes * Bytes))), Bytes] (Studio.Doc::cat (cons (project doc path) (cons (Studio.Doc::lit ": an xs:") (cons (project head kind) (cons (Studio.Doc::lit " with no schemaLocation names no document whose components could be read (XSD 1.1 Part 1, section 4.2.6.2)") (nil Bytes)))))))
        | Option.some location => (match (resolved (project doc url) location)
          | Result.error reason => (ctor Result.error [(List (Bytes * (Bytes * Bytes))), Bytes] (Studio.Doc::cat (cons (project doc path) (cons (Studio.Doc::lit ": ") (cons reason (nil Bytes))))))
          | Result.ok url => (match (artifactOf sources url)
            | Option.none => (ctor Result.error [(List (Bytes * (Bytes * Bytes))), Bytes] (Studio.Doc::cat (cons (project doc path) (cons (Studio.Doc::lit ": the schema document ") (cons url (cons (Studio.Doc::lit " is not a bound authority, so its components cannot be read; stage its binding first (XSD 1.1 Part 1, section 4.2.6.2)") (nil Bytes)))))))
            | Option.some artifact => (ctor Result.ok [(List (Bytes * (Bytes * Bytes))), Bytes] (cons (artifact, ((project head kind), (if (isKind head "include") then (project doc target) else (attrOr head "namespace" "")))) later))))) else (ctor Result.ok [(List (Bytes * (Bytes * Bytes))), Bytes] later))))),
  recursive_argument: items,
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: lookupFile,
  parameters: [files : (List (Bytes * (Option Bytes))), path : Bytes],
  result: (Option (Option Bytes)),
  body: (match files
    | List.nil => (ctor Option.none [(Option Bytes)])
    | List.cons head rest => (if (Studio.Doc::same (first head) path) then (ctor Option.some [(Option Bytes)] (second head)) else (lookupFile rest path))),
  recursive_argument: files,
  executable: true
}

definition{
  name: docPaths,
  parameters: [docs : (List Document)],
  result: (List Bytes),
  body: (match docs
    | List.nil => (nil Bytes)
    | List.cons head rest => (cons (project head path) (docPaths rest))),
  recursive_argument: docs,
  executable: true
}

definition{
  name: reverseDocs,
  parameters: [docs : (List Document), done : (List Document)],
  result: (List Document),
  body: (match docs
    | List.nil => done
    | List.cons head rest => (reverseDocs rest (cons head done))),
  recursive_argument: docs,
  executable: true
}

definition{
  name: appendQueue,
  parameters: [left : (List (Bytes * (Bytes * Bytes))), right : (List (Bytes * (Bytes * Bytes)))],
  result: (List (Bytes * (Bytes * Bytes))),
  body: (match left
    | List.nil => right
    | List.cons head rest => (cons head (appendQueue rest right))),
  recursive_argument: left,
  executable: true
}

definition{
  name: expectation,
  parameters: [doc : Document, kind : Bytes, space : Bytes],
  result: (Option Bytes),
  body: (if (Studio.Doc::same kind (Studio.Doc::lit "root")) then (ctor Option.none [Bytes]) else (if (Studio.Doc::same (project doc target) space) then (ctor Option.none [Bytes]) else (ctor Option.some [Bytes] (Studio.Doc::cat (cons (project doc path) (cons (Studio.Doc::lit " has the target namespace ") (cons (project doc target) (cons (Studio.Doc::lit ", but the xs:") (cons kind (cons (Studio.Doc::lit " that names it requires ") (cons space (cons (if (Studio.Doc::same kind (Studio.Doc::lit "include")) then (Studio.Doc::lit "; an included document without a target namespace is not adopted into the including one (XSD 1.1 Part 1, section 4.2.3)") else (Studio.Doc::lit " (XSD 1.1 Part 1, section 4.2.6.2)")) (nil Bytes))))))))))))),
  executable: true
}

definition{
  name: assemble,
  parameters: [fuel : Nat, files : (List (Bytes * (Option Bytes))), sources : (List (Bytes * Bytes)), queue : (List (Bytes * (Bytes * Bytes))), docs : (List Document), needs : (List Bytes)],
  result: Outcome,
  body: (match fuel
    | Nat.zero => (ctor Outcome.refuses (Studio.Doc::lit "the schema documents import and include more documents than the bound authorities hold"))
    | Nat.succ rest => (match queue
      | List.nil => (match needs
        | List.nil => (match (translateSet (reverseDocs docs (nil Document)))
          | Result.error reason => (ctor Outcome.refuses reason)
          | Result.ok translated => (ctor Outcome.translated (first translated) (second translated)))
        | List.cons head more => (ctor Outcome.needs needs))
      | List.cons head more => (let path : Bytes := (first head); (if ((Studio.Doc::memberBytes path (docPaths docs)) || (Studio.Doc::memberBytes path needs)) then (assemble rest files sources more docs needs) else (match (lookupFile files path)
        | Option.none => (assemble rest files sources more docs (cons path needs))
        | Option.some found => (match found
          | Option.none => (ctor Outcome.refuses (Studio.Doc::cat (cons (Studio.Doc::lit "the schema document ") (cons path (cons (Studio.Doc::lit " is not in the repository") (nil Bytes))))))
          | Option.some raw => (match (parseDocument path raw (urlOf sources path))
            | Result.error reason => (ctor Outcome.refuses reason)
            | Result.ok doc => (match (expectation doc (first (second head)) (second (second head)))
              | Option.some reason => (ctor Outcome.refuses reason)
              | Option.none => (match (nodes (project doc content) (project doc scope))
                | Result.error reason => (ctor Outcome.refuses reason)
                | Result.ok items => (match (referencesIn items doc sources)
                  | Result.error reason => (ctor Outcome.refuses reason)
                  | Result.ok found2 => (assemble rest files sources (appendQueue more found2) (cons doc docs) needs))))))))))),
  recursive_argument: fuel,
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: readSchemaSet,
  parameters: [files : (List (Bytes * (Option Bytes))), authorities : Bytes, path : Bytes],
  result: Outcome,
  body: (let sources : (List (Bytes * Bytes)) := (boundSources authorities); (assemble ((prim length sources : Nat) + 2) files sources (cons (path, ((Studio.Doc::lit "root"), 0x)) (nil (Bytes * (Bytes * Bytes)))) (nil Document) (nil Bytes))),
  axioms: [Quot.sound, propext],
  executable: true
}
