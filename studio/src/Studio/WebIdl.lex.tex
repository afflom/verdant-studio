  result: Nat,
  body: (match items
    | List.nil => 0
    | List.cons head rest => (if (Studio.Doc::same (first (second head)) 0x) then (blockedCount rest) else ((blockedCount rest) + 1))),
  recursive_argument: items,
  executable: true

    | Option.none => (closure rest others done lookup)
