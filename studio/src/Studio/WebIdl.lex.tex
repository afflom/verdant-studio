    | Option.some newline => (tokensAt rest source (newline + 1) acc)) else (if (Studio.Xml::startsAt source position "/*") then (match (Studio.Xml::findFrom source (prim utf8_encode "*/" : Bytes) (position + 2))
    | Option.none => (ctor Result.error [(List IdlToken), Bytes] (failure "a comment is not closed (Web IDL, IDL grammar: comment)" position))
    | Option.some close => (tokensAt rest source (close + 2) acc)) else (if (byte =? 34) then (match (Studio.Xml::findFrom source 0x22 (position + 1))

    | Nat.zero => (ctor Result.error [(List IdlToken), Bytes] (Studio.Doc::lit "brackets nest deeper than the input is long"))
