definition{
  name: declarations,
  parameters: [attrs : Studio.Xml::Xml, scope : (List (Bytes * Bytes))],
  result: (Result (List (Bytes * Bytes)) Bytes),
  body: (match attrs
    | Studio.Xml::Xml.XAttr label value rest => (match (declarations rest scope)
      | Result.error reason => (failScope reason)
      | Result.ok inner => (if (Studio.Doc::same label (Studio.Doc::lit "xmlns")) then (if ((Studio.Doc::same value (xmlNamespace)) || (Studio.Doc::same value (xmlnsNamespace))) then (failScope (Studio.Doc::lit "the default namespace must not be the xml or xmlns namespace name (Namespaces in XML 1.0, section 3, NSC: Reserved Prefixes and Namespace Names)")) else (ctor Result.ok [(List (Bytes * Bytes)), Bytes] (cons (0x, value) inner))) else (if (LexLeanStd.Text::startsWith label (Studio.Doc::lit "xmlns:")) then (let declared : Bytes := (LexLeanStd.Text::sliceOr label 6 (prim subtract (prim length label : Nat) 6 : Nat)); (if (Studio.Doc::same declared (Studio.Doc::lit "xmlns")) then (failScope (Studio.Doc::lit "the prefix xmlns must not be declared (Namespaces in XML 1.0, section 3, NSC: Reserved Prefixes and Namespace Names)")) else (if ((Studio.Doc::same declared (Studio.Doc::lit "xml")) && (if (Studio.Doc::same value (xmlNamespace)) then false else true)) then (failScope (Studio.Doc::lit "the prefix xml may be bound only to its own namespace name (Namespaces in XML 1.0, section 3, NSC: Reserved Prefixes and Namespace Names)")) else (if (((Studio.Doc::same value (xmlNamespace)) || (Studio.Doc::same value (xmlnsNamespace))) && (if (Studio.Doc::same declared (Studio.Doc::lit "xml")) then false else true)) then (failScope (Studio.Doc::lit "the xml and xmlns namespace names must not be bound to another prefix (Namespaces in XML 1.0, section 3, NSC: Reserved Prefixes and Namespace Names)")) else (if (Studio.Doc::same value 0x) then (failScope (Studio.Doc::lit "a prefix must not be undeclared (Namespaces in XML 1.0, section 3, NSC: No Prefix Undeclaring)")) else (ctor Result.ok [(List (Bytes * Bytes)), Bytes] (cons (declared, value) inner))))))) else (ctor Result.ok [(List (Bytes * Bytes)), Bytes] inner))))
    | Studio.Xml::Xml.XElem label inner content => (ctor Result.ok [(List (Bytes * Bytes)), Bytes] scope)
    | Studio.Xml::Xml.XText text => (ctor Result.ok [(List (Bytes * Bytes)), Bytes] scope)
    | Studio.Xml::Xml.XItem head rest => (ctor Result.ok [(List (Bytes * Bytes)), Bytes] scope)
    | Studio.Xml::Xml.XNil => (ctor Result.ok [(List (Bytes * Bytes)), Bytes] scope)),
  recursive_argument: attrs,
  executable: true
}

definition{
  name: expand,
  parameters: [scope : (List (Bytes * Bytes)), qname : Bytes],
  result: (Result (Bytes * Bytes) Bytes),
  body: (let prefix : Bytes := (prefixOf qname); (if (Studio.Doc::same prefix 0x) then (match (lookup scope 0x)
    | Option.none => (ctor Result.ok [(Bytes * Bytes), Bytes] (0x, qname))
    | Option.some uri => (ctor Result.ok [(Bytes * Bytes), Bytes] (uri, qname))) else (if (Studio.Doc::same prefix (Studio.Doc::lit "xml")) then (ctor Result.ok [(Bytes * Bytes), Bytes] ((xmlNamespace), (localOf qname))) else (match (lookup scope prefix)
    | Option.none => (ctor Result.error [(Bytes * Bytes), Bytes] (Studio.Doc::cat (cons (Studio.Doc::lit "the prefix ") (cons prefix (cons (Studio.Doc::lit " is not declared (Namespaces in XML 1.0, section 5, NSC: Prefix Declared)") (nil Bytes))))))
    | Option.some uri => (ctor Result.ok [(Bytes * Bytes), Bytes] (uri, (localOf qname))))))),
  executable: true
}

definition{
  name: builtinParent,
  parameters: [local : Bytes],
  result: (Option Bytes),
  body: (match (Studio.Xml::findFrom local (Studio.Doc::lit " ") 0)
    | Option.some space => (ctor Option.none [Bytes])
    | Option.none => (match (Studio.Xml::findFrom local (Studio.Doc::lit ":") 0)
      | Option.some colon => (ctor Option.none [Bytes])
      | Option.none => (let needle : Bytes := (Studio.Doc::cat (cons (Studio.Doc::lit " ") (cons local (cons (Studio.Doc::lit ":") (nil Bytes))))); (match (Studio.Xml::findFrom (parentTable) needle 0)
        | Option.none => (ctor Option.none [Bytes])
        | Option.some found => (let start : Nat := (found + (prim length needle : Nat)); (match (Studio.Xml::findFrom (parentTable) (Studio.Doc::lit " ") start)
          | Option.none => (ctor Option.none [Bytes])
          | Option.some stop => (ctor Option.some [Bytes] (LexLeanStd.Text::sliceOr (parentTable) start (prim subtract stop start : Nat))))))))),
  executable: true
}

definition{
  name: builtinType,
  parameters: [local : Bytes, fuel : Nat],
  result: (Result Bytes Bytes),
  body: (match fuel
    | Nat.zero => (failText (Studio.Doc::lit "the built-in datatype hierarchy is deeper than XSD 1.1 Part 2 section 3 defines"))
    | Nat.succ rest => (if (Studio.Xml::member local (cons (Studio.Doc::lit "NMTOKENS") (cons (Studio.Doc::lit "IDREFS") (cons (Studio.Doc::lit "ENTITIES") (nil Bytes))))) then (ctor Result.ok [Bytes, Bytes] (Studio.Doc::lit "(List Bytes)")) else (if (Studio.Doc::same local (Studio.Doc::lit "nonNegativeInteger")) then (ctor Result.ok [Bytes, Bytes] (Studio.Doc::lit "Nat")) else (if (Studio.Doc::same local (Studio.Doc::lit "integer")) then (ctor Result.ok [Bytes, Bytes] (Studio.Doc::lit "Int")) else (if (Studio.Doc::same local (Studio.Doc::lit "decimal")) then (ctor Result.ok [Bytes, Bytes] (Studio.Schema::numberType)) else (if (Studio.Doc::same local (Studio.Doc::lit "boolean")) then (ctor Result.ok [Bytes, Bytes] (Studio.Doc::lit "Bool")) else (if (Studio.Xml::member local (cons (Studio.Doc::lit "float") (cons (Studio.Doc::lit "double") (nil Bytes)))) then (failText (Studio.Doc::cat (cons (Studio.Doc::lit "xs:") (cons local (cons (Studio.Doc::lit " is not translated, because its value space holds positive and negative zero, the infinities and NaN, which no exact LexLean number type holds (XSD 1.1 Part 2, sections 3.3.4 and 3.3.5)") (nil Bytes)))))) else (if (Studio.Xml::member local (cons (Studio.Doc::lit "anySimpleType") (cons (Studio.Doc::lit "anyAtomicType") (nil Bytes)))) then (failText (Studio.Doc::cat (cons (Studio.Doc::lit "xs:") (cons local (cons (Studio.Doc::lit " is not translated, because it is the union of every simple value space and has no one LexLean type (XSD 1.1 Part 2, sections 3.2.1 and 3.2.2)") (nil Bytes)))))) else (match (builtinParent local)
      | Option.none => (failText (Studio.Doc::cat (cons (Studio.Doc::lit "xs:") (cons local (cons (Studio.Doc::lit " is not a built-in datatype of XSD 1.1 Part 2") (nil Bytes))))))
      | Option.some parent => (if (Studio.Doc::same parent (Studio.Doc::lit "anyAtomicType")) then (ctor Result.ok [Bytes, Bytes] (Studio.Doc::lit "Bytes")) else (builtinType parent rest))))))))))),
  recursive_argument: fuel,
  executable: true
}

  axioms: [Quot.sound, propext],
  executable: true

        | Studio.Xml::Xml.XNil => (ctor Result.ok [(List Studio.Schema::Value), Bytes] later)))
    | Studio.Xml::Xml.XElem label attrs inner => (ctor Result.ok [(List Studio.Schema::Value), Bytes] (nil Studio.Schema::Value))
    | Studio.Xml::Xml.XAttr label value more => (ctor Result.ok [(List Studio.Schema::Value), Bytes] (nil Studio.Schema::Value))
    | Studio.Xml::Xml.XText text => (ctor Result.ok [(List Studio.Schema::Value), Bytes] (nil Studio.Schema::Value))

definition{
  name: derivation,
  parameters: [content : Studio.Xml::Xml, scope : (List (Bytes * Bytes))],
  result: (Result (Option Derived) Bytes),
  body: (match content
    | Studio.Xml::Xml.XItem head rest => (match head
      | Studio.Xml::Xml.XElem label attrs inner => (match (declarations attrs scope)
        | Result.error reason => (failDerived reason)
        | Result.ok local => (match (expand local label)
          | Result.error reason => (failDerived reason)
          | Result.ok expanded => (if (Studio.Xml::member (second expanded) (cons (Studio.Doc::lit "restriction") (cons (Studio.Doc::lit "list") (cons (Studio.Doc::lit "union") (nil Bytes))))) then (if (Studio.Doc::same (first expanded) (xsdNamespace)) then (ctor Result.ok [(Option Derived), Bytes] (ctor Option.some [Derived] (record Derived {kind := (second expanded), attrs := attrs, inner := inner, bindings := local}))) else (derivation rest scope)) else (derivation rest scope))))
      | Studio.Xml::Xml.XAttr label value more => (derivation rest scope)
      | Studio.Xml::Xml.XText text => (derivation rest scope)
      | Studio.Xml::Xml.XItem first1 more => (derivation rest scope)
      | Studio.Xml::Xml.XNil => (derivation rest scope))
    | Studio.Xml::Xml.XElem label attrs inner => (ctor Result.ok [(Option Derived), Bytes] (ctor Option.none [Derived]))
    | Studio.Xml::Xml.XAttr label value more => (ctor Result.ok [(Option Derived), Bytes] (ctor Option.none [Derived]))
    | Studio.Xml::Xml.XText text => (ctor Result.ok [(Option Derived), Bytes] (ctor Option.none [Derived]))
    | Studio.Xml::Xml.XNil => (ctor Result.ok [(Option Derived), Bytes] (ctor Option.none [Derived]))),
  recursive_argument: content,
  executable: true
}

definition{
  name: simpleType,
  parameters: [named : Bytes, content : Studio.Xml::Xml, scope : (List (Bytes * Bytes)), target : Bytes],
  result: (Result Carried Bytes),
  body: (let typeName : Bytes := (Studio.Schema::pascal named); (match (derivation content scope)
    | Result.error reason => (failCarried (located reason named))
    | Result.ok found => (match found
      | Option.none => (failCarried (located (Studio.Doc::lit "a simpleType must have a restriction, list or union child (XSD 1.1 Part 2, section 4.1.2)") named))
      | Option.some chosen => (let kind : Bytes := (project chosen kind); (let attrs : Studio.Xml::Xml := (project chosen attrs); (let inner : Studio.Xml::Xml := (project chosen inner); (let local : (List (Bytes * Bytes)) := (project chosen bindings); (if (Studio.Doc::same kind (Studio.Doc::lit "union")) then (failCarried (located (Studio.Doc::lit "a union is not translated, because a literal can be valid for more than one member type, so no LexLean inductive keeps its members disjoint (XSD 1.1 Part 2, section 2.4.1.3)") named)) else (if (Studio.Doc::same kind (Studio.Doc::lit "list")) then (match (attrValue attrs (Studio.Doc::lit "itemType"))
        | Option.none => (failCarried (located (Studio.Doc::lit "a list with an anonymous item type is not translated yet") named))
        | Option.some item => (match (typeOf local target item)
          | Result.error reason => (failCarried (located reason named))
          | Result.ok carried => (if (LexLeanStd.Text::startsWith (project carried text) (Studio.Doc::lit "(List ")) then (failCarried (located (Studio.Doc::lit "the item type of a list must not itself be a list (XSD 1.1 Part 2, section 4.1.5)") named)) else (ctor Result.ok [Carried, Bytes] (record Carried {text := (valueStructure typeName (Studio.Doc::cat (cons (Studio.Doc::lit "(List ") (cons (project carried text) (cons (Studio.Doc::lit ")") (nil Bytes)))))), refs := (project carried refs)}))))) else (match (attrValue attrs (Studio.Doc::lit "base"))
        | Option.none => (failCarried (located (Studio.Doc::lit "a restriction with an anonymous base type is not translated yet") named))
        | Option.some base => (match (typeOf local target base)
          | Result.error reason => (failCarried (located reason named))
          | Result.ok carried => (match (enumerations inner local)
            | Result.error reason => (failCarried (located reason named))
            | Result.ok listed => (let values : (List Studio.Schema::Value) := (distinctValues listed (nil Bytes)); (match values
              | List.nil => (ctor Result.ok [Carried, Bytes] (record Carried {text := (valueStructure typeName (project carried text)), refs := (project carried refs)}))
              | List.cons head others => (match (Studio.Schema::constructorsWith (Studio.Schema::collides (Studio.Schema::strings values)) values (nil Bytes))
                | Result.error reason => (failCarried (located reason named))
                | Result.ok constructors => (ctor Result.ok [Carried, Bytes] (record Carried {text := (Studio.Schema::inductiveText typeName constructors), refs := (nil Bytes)}))))))))))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: simpleTypes,
  parameters: [content : Studio.Xml::Xml, scope : (List (Bytes * Bytes)), target : Bytes],
  result: (Result (List (Bytes * ((List Bytes) * Bytes))) Bytes),
  body: (match content
    | Studio.Xml::Xml.XItem head rest => (match (simpleTypes rest scope target)
      | Result.error reason => (failTypes reason)
      | Result.ok later => (match head
        | Studio.Xml::Xml.XElem label attrs inner => (match (declarations attrs scope)
          | Result.error reason => (failTypes reason)
          | Result.ok local => (match (expand local label)
            | Result.error reason => (failTypes reason)
            | Result.ok expanded => (if (isXsd expanded "simpleType") then (match (attrValue attrs (Studio.Doc::lit "name"))
              | Option.none => (failTypes (Studio.Doc::lit "a top-level simpleType must have a name (XSD 1.1 Part 2, section 4.1.2)"))
              | Option.some named => (match (simpleType named inner local target)
                | Result.error reason => (failTypes reason)
                | Result.ok translated => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] (cons ((Studio.Schema::pascal named), ((project translated refs), (project translated text))) later)))) else (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] later))))
        | Studio.Xml::Xml.XAttr label value more => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] later)
        | Studio.Xml::Xml.XText text => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] later)
        | Studio.Xml::Xml.XItem first1 more => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] later)
        | Studio.Xml::Xml.XNil => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] later)))
    | Studio.Xml::Xml.XElem label attrs inner => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] (nil (Bytes * ((List Bytes) * Bytes))))
    | Studio.Xml::Xml.XAttr label value more => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] (nil (Bytes * ((List Bytes) * Bytes))))
    | Studio.Xml::Xml.XText text => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] (nil (Bytes * ((List Bytes) * Bytes))))
    | Studio.Xml::Xml.XNil => (ctor Result.ok [(List (Bytes * ((List Bytes) * Bytes))), Bytes] (nil (Bytes * ((List Bytes) * Bytes))))),
  recursive_argument: content,
  axioms: [Quot.sound, propext],
  executable: true
}

}

definition{
  name: undefinedReference,
  parameters: [types : (List (Bytes * ((List Bytes) * Bytes))), defined : (List Bytes)],
  result: (Option Bytes),
  body: (match types
    | List.nil => (ctor Option.none [Bytes])
    | List.cons head rest => (match (missingName (first (second head)) defined)
      | Option.some absent => (ctor Option.some [Bytes] absent)
      | Option.none => (undefinedReference rest defined))),
  recursive_argument: types,
  executable: true
}

definition{
  name: pickReady,
  parameters: [pending : (List (Bytes * ((List Bytes) * Bytes))), emitted : (List Bytes)],
  result: (Option ((Bytes * Bytes) * (List (Bytes * ((List Bytes) * Bytes))))),
  body: (match pending
    | List.nil => (ctor Option.none [((Bytes * Bytes) * (List (Bytes * ((List Bytes) * Bytes))))])
    | List.cons head rest => (match (missingName (first (second head)) emitted)
      | Option.none => (ctor Option.some [((Bytes * Bytes) * (List (Bytes * ((List Bytes) * Bytes))))] (((first head), (second (second head))), rest))
      | Option.some absent => (match (pickReady rest emitted)
        | Option.none => (ctor Option.none [((Bytes * Bytes) * (List (Bytes * ((List Bytes) * Bytes))))])
        | Option.some picked => (ctor Option.some [((Bytes * Bytes) * (List (Bytes * ((List Bytes) * Bytes))))] ((first picked), (cons head (second picked))))))),
  recursive_argument: pending,
  executable: true
}

definition{
  name: ordered,
  parameters: [fuel : Nat, pending : (List (Bytes * ((List Bytes) * Bytes))), emitted : (List Bytes), out : (List Bytes)],
  result: (Result (List Bytes) Bytes),
  body: (match fuel
    | Nat.zero => (match pending
      | List.nil => (ctor Result.ok [(List Bytes), Bytes] out)
      | List.cons waiting others => (failTexts (Studio.Doc::lit "the simple types derive from each other in a cycle (XSD 1.1 Part 2, section 4.1.5)")))
    | Nat.succ rest => (match pending
      | List.nil => (ctor Result.ok [(List Bytes), Bytes] out)
      | List.cons waiting others => (match (pickReady pending emitted)
        | Option.none => (failTexts (Studio.Doc::lit "the simple types derive from each other in a cycle (XSD 1.1 Part 2, section 4.1.5)"))
        | Option.some picked => (ordered rest (second picked) (cons (first (first picked)) emitted) (cons (second (first picked)) out))))),
  recursive_argument: fuel,
  executable: true
}

definition{
  name: reversed,
  parameters: [items : (List Bytes), done : (List Bytes)],
  result: (List Bytes),
  body: (match items
    | List.nil => done
    | List.cons head rest => (reversed rest (cons head done))),
  recursive_argument: items,
  executable: true
}

definition{
  name: translateXsd,
  parameters: [raw : Bytes],
  result: (Result (Nat * Bytes) Bytes),
  body: (match (Studio.Xml::readXml raw)
    | Result.error reason => (failTranslation (Studio.Doc::cat (cons (Studio.Doc::lit "not well-formed XML: ") (cons reason (nil Bytes)))))
    | Result.ok root => (match root
      | Studio.Xml::Xml.XElem label attrs content => (match (declarations attrs (nil (Bytes * Bytes)))
        | Result.error reason => (failTranslation reason)
        | Result.ok scope => (match (expand scope label)
          | Result.error reason => (failTranslation reason)
          | Result.ok expanded => (if (isXsd expanded "schema") then (let target : Bytes := (match (attrValue attrs (Studio.Doc::lit "targetNamespace"))
            | Option.none => 0x
            | Option.some uri => uri); (match (simpleTypes content scope target)
            | Result.error reason => (failTranslation reason)
            | Result.ok types => (let names : (List Bytes) := (typeNames types); (match (duplicateName names)
              | Option.some twice => (failTranslation (Studio.Doc::cat (cons (Studio.Doc::lit "two simple types derive the type name ") (cons twice (nil Bytes)))))
              | Option.none => (match (undefinedReference types names)
                | Option.some absent => (failTranslation (Studio.Doc::cat (cons (Studio.Doc::lit "the type ") (cons absent (cons (Studio.Doc::lit " is not a simple type of this schema document") (nil Bytes))))))
                | Option.none => (match (ordered ((prim length names : Nat) + 1) types (nil Bytes) (nil Bytes))
                  | Result.error reason => (failTranslation reason)
                  | Result.ok texts => (ctor Result.ok [(Nat * Bytes), Bytes] ((prim length names : Nat), (Studio.Schema::joinWith (reversed texts (nil Bytes)) (Studio.Doc::lit "\n")))))))))) else (failTranslation (Studio.Doc::lit "the root element must be xs:schema in the XML Schema namespace (XSD 1.1 Part 1, section 3.17.2)")))))
      | Studio.Xml::Xml.XAttr label value more => (failTranslation (Studio.Doc::lit "the document has no root element"))
      | Studio.Xml::Xml.XText text => (failTranslation (Studio.Doc::lit "the document has no root element"))
      | Studio.Xml::Xml.XItem head rest => (failTranslation (Studio.Doc::lit "the document has no root element"))
      | Studio.Xml::Xml.XNil => (failTranslation (Studio.Doc::lit "the document has no root element")))),
  axioms: [Quot.sound, propext],
  executable: true
}
