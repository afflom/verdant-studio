definition{
  name: writeNext,
  parameters: [session : Session],
  result: (Session * LexLeanStd.Host::Request),
  body: (match (project session writes)
    | List.nil => (quit session (0 : UInt8) (project session message) 0x)
    | List.cons head rest => ((withPhase session 5 (first head)), (ctor LexLeanStd.Host::Request.createDir (VerdantStudioCli.Studio.Doc::text (parentOf (first head)))))),

  executable: true

definition{
  name: advance,
  parameters: [session : Session],
  result: (Session * LexLeanStd.Host::Request),
  body: (match (VerdantStudioCli.Studio.Ops::admitPlan (project session args) (project session seen))
    | VerdantStudioCli.Studio.Ops::Plan.reads paths => (match paths
      | List.nil => (quit session (1 : UInt8) 0x (VerdantStudioCli.Studio.Doc::lit "internal: an empty read plan\n"))
      | List.cons head rest => ((withPending (withPhase session 1 head) rest), (ctor LexLeanStd.Host::Request.read (VerdantStudioCli.Studio.Doc::text head))))
    | VerdantStudioCli.Studio.Ops::Plan.lists paths => (match paths
      | List.nil => (quit session (1 : UInt8) 0x (VerdantStudioCli.Studio.Doc::lit "internal: an empty list plan\n"))
      | List.cons head rest => ((withPhase session 2 head), (ctor LexLeanStd.Host::Request.list (VerdantStudioCli.Studio.Doc::text head))))
    | VerdantStudioCli.Studio.Ops::Plan.run program arguments tag => ((record Session {pending := (project session pending), args := (project session args), environment := (project session environment), seen := (project session seen), phase := 3, current := tag, program := program, arguments := arguments, handle := 0, writes := (project session writes), message := (project session message)}), (ctor LexLeanStd.Host::Request.createDir "target"))
    | VerdantStudioCli.Studio.Ops::Plan.done writes message => (writeNext (record Session {pending := (project session pending), args := (project session args), environment := (project session environment), seen := (project session seen), phase := 5, current := 0x, program := (project session program), arguments := (project session arguments), handle := (project session handle), writes := writes, message := message}))
    | VerdantStudioCli.Studio.Ops::Plan.refuse reason => (quit session (1 : UInt8) 0x reason)),
  axioms: [Quot.sound, propext],
  executable: true
}

  name: recordRead,
  parameters: [session : Session, found : (Option Bytes)],

  body: (let seen : VerdantStudioCli.Studio.Ops::Seen := (project session seen); (let updated : Session := (withSeen session (VerdantStudioCli.Studio.Ops::readInto seen (project session current) found)); (match (project session pending)
    | List.nil => (advance updated)
    | List.cons head rest => ((withPending (withPhase updated 1 head) rest), (ctor LexLeanStd.Host::Request.read (VerdantStudioCli.Studio.Doc::text head)))))),
  axioms: [Quot.sound, propext],
  executable: true
}

  parameters: [finished : (List (Nat * (Option Int32))), handle : Nat],
  result: (Option (Option Int32)),
  body: (match finished
    | List.nil => (ctor Option.none [(Option Int32)])
    | List.cons head rest => (if ((first head) =? handle) then (ctor Option.some [(Option Int32)] (second head)) else (exitOf rest handle))),

  executable: true

definition{
  name: run,
  parameters: [previous : (Option Session), event : LexLeanStd.Host::Event],
  result: (Session * LexLeanStd.Host::Request),
  body: (let session : Session := (match previous
      | Option.none => (fresh)
      | Option.some known => known); (match event
    | LexLeanStd.Host::Event.start start => (advance (record Session {pending := (nil Bytes), args := (project start arguments), environment := (project start environment), seen := (emptySeen), phase := 0, current := 0x, program := 0x, arguments := (nil Bytes), handle := 0, writes := (nil (Bytes * Bytes)), message := 0x}))
    | LexLeanStd.Host::Event.read result => (if ((project session phase) =? 7) then ((record Session {pending := (project session pending), args := (project session args), environment := (project session environment), seen := (project session seen), phase := 8, current := (project session current), program := (project session program), arguments := (project session arguments), handle := (project session handle), writes := (project session writes), message := (match result
        | Result.ok text => text
        | Result.error reason => 0x)}), (ctor LexLeanStd.Host::Request.read (VerdantStudioCli.Studio.Doc::text (childErr)))) else (if ((project session phase) =? 8) then (quit session (1 : UInt8) (project session message) (VerdantStudioCli.Studio.Doc::cat (cons (match result
        | Result.ok text => text
        | Result.error reason => 0x) (cons (VerdantStudioCli.Studio.Doc::lit "command failed: ") (cons (project session current) (cons 0x0a (nil Bytes))))))) else (match result
        | Result.ok text => (recordRead session (ctor Option.some [Bytes] text))
        | Result.error reason => (recordRead session (ctor Option.none [Bytes])))))
    | LexLeanStd.Host::Event.listed result => (match result
      | Result.ok names => (recordListing session names)
      | Result.error reason => (recordListing session (nil Bytes)))
    | LexLeanStd.Host::Event.finished result => (match result
      | Result.error reason => (failed session (VerdantStudioCli.Studio.Doc::lit "could not write"))
      | Result.ok done => (if ((project session phase) =? 3) then (spawnChild session) else (if ((project session phase) =? 5) then (match (project session writes)
          | List.nil => (writeNext session)
          | List.cons head rest => ((withPhase session 6 (first head)), (ctor LexLeanStd.Host::Request.write (VerdantStudioCli.Studio.Doc::text (first head)) (second head)))) else (match (project session writes)
          | List.nil => (writeNext session)
          | List.cons head rest => (writeNext (record Session {pending := (project session pending), args := (project session args), environment := (project session environment), seen := (project session seen), phase := 5, current := 0x, program := (project session program), arguments := (project session arguments), handle := (project session handle), writes := rest, message := (project session message)}))))))
    | LexLeanStd.Host::Event.spawned result => (match result
      | Result.error reason => (failed session (VerdantStudioCli.Studio.Doc::lit "could not start"))
      | Result.ok started => (waitAgain (record Session {pending := (project session pending), args := (project session args), environment := (project session environment), seen := (project session seen), phase := 4, current := (project session current), program := (project session program), arguments := (project session arguments), handle := (first started), writes := (project session writes), message := (project session message)})))
    | LexLeanStd.Host::Event.waited result => (match result
      | Result.error reason => (failed session (VerdantStudioCli.Studio.Doc::lit "could not wait for"))
      | Result.ok waited => (match (exitOf (project waited finished) (project session handle))
        | Option.none => (waitAgain session)
        | Option.some code => (if (succeeded code) then (recordRan session) else ((withPhase session 7 (project session current)), (ctor LexLeanStd.Host::Request.read (VerdantStudioCli.Studio.Doc::text (childOut)))))))
    | LexLeanStd.Host::Event.stat result => (failed session (VerdantStudioCli.Studio.Doc::lit "internal: unexpected stat"))
    | LexLeanStd.Host::Event.canonical result => (failed session (VerdantStudioCli.Studio.Doc::lit "internal: unexpected canonical")))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, files, overflow, process, recursion], targets: [rust-std]}
}
